# ===================================================
# WORDPRESS CORE & SECURITY RULES
# One group for every crawler, so the AI search and citation bots below follow the same
# exclusions (a crawler obeys only the most specific group that names it).
# ===================================================
User-agent: *
# ALLOW AI SEARCH & CITATION BOTS (GEO Strategy)
# Allows AI models to search, read, and attribute articles
User-agent: OAI-SearchBot
User-agent: ChatGPT-User
User-agent: PerplexityBot
User-agent: Claude-SearchBot
User-agent: Claude-Web
User-agent: GPTBot
User-agent: ClaudeBot
User-agent: Google-Extended
Disallow: /wp-admin/
Allow: /wp-admin/admin-ajax.php
Disallow: /readme.html
Disallow: /license.txt
# Personal pages (also noindex): accounts, reading dashboard
Disallow: /account/
Disallow: /dashboard/
# Internal search results and comment-reply duplicates. Other query strings (e.g. utm_ tags) stay crawlable:
# their canonical tag points at the clean URL, and CSS/JS with ?ver= must stay fetchable for rendering.
Disallow: /?s=
Disallow: /*?s=
Disallow: /*&s=
Disallow: /search/
Disallow: /*?replytocom=
# Signature checks use query strings (/verify/?sig=...); the page itself is public
Disallow: /verify/?
Allow: /signed/
Allow: /.well-known/pqsig-keys.json
